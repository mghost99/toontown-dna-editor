cmake_minimum_required(VERSION 3.20)
project(dna-editor VERSION 0.0.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Panda3D C++ SDK (headers + libs). Default: the local source build at ~/panda3d/built.
if(WIN32)
  set(_p3default "C:/Panda3D-1.11.0-x64")
else()
  set(_p3default "$ENV{HOME}/panda3d/built")
endif()
set(PANDA3D_ROOT "${_p3default}" CACHE PATH "Panda3D SDK root (contains include/ and lib/)")
find_path(PANDA3D_INCLUDE_DIR pandaFramework.h PATHS ${PANDA3D_ROOT}/include /usr/include/panda3d NO_DEFAULT_PATH)
find_library(P3_PANDA NAMES panda libpanda PATHS ${PANDA3D_ROOT}/lib NO_DEFAULT_PATH)
if(NOT PANDA3D_INCLUDE_DIR OR NOT P3_PANDA)
  message(FATAL_ERROR "Panda3D SDK not found; set -DPANDA3D_ROOT=/path/to/panda3d/built")
endif()
get_filename_component(PANDA3D_LIB_DIR ${P3_PANDA} DIRECTORY)
add_library(panda3d INTERFACE)
target_include_directories(panda3d SYSTEM INTERFACE ${PANDA3D_INCLUDE_DIR} /usr/include/eigen3)
target_link_directories(panda3d INTERFACE ${PANDA3D_LIB_DIR})
if(WIN32)
  target_link_libraries(panda3d INTERFACE libp3framework libpanda libpandaexpress libp3dtoolconfig libp3dtool)
  target_compile_definitions(panda3d INTERFACE _USE_MATH_DEFINES NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(panda3d INTERFACE /utf-8 /bigobj /wd4244 /wd4267 /wd4305 /wd4996)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
else()
  target_link_libraries(panda3d INTERFACE p3framework panda pandaexpress p3dtoolconfig p3dtool pthread)
endif()
set(CMAKE_BUILD_RPATH ${PANDA3D_LIB_DIR})
set(CMAKE_INSTALL_RPATH ${PANDA3D_LIB_DIR})

add_subdirectory(third_party)
add_subdirectory(src/core)
add_subdirectory(src/dna)
add_subdirectory(src/commands)
add_subdirectory(src/assets)
add_subdirectory(src/render)
add_subdirectory(src/ui)
add_subdirectory(src/platform)
add_subdirectory(app)
option(ED_BUILD_TESTS "Build the engine-free dna module tests" ON)
if(ED_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
